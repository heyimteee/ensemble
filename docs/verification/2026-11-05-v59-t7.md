Doc link check command (bash; checks changed markdown files):

```sh
files=$(git ls-files -m -o --exclude-standard -- '*.md')
for f in $files; do
  while IFS= read -r link; do
    case "$link" in http*|mailto:*|"") continue;; esac
    clean=${link%%#*}; [ -z "$clean" ] && continue
    [ -e "$(dirname "$f")/$clean" ] || echo "BROKEN: $f -> $link"
  done < <(grep -oE '\]\([^)]+' "$f" | cut -c3-)
done
```
